# Pinned to compatible releases (~=) rather than left floating: these are the
# Third-party dependencies are installed fresh on every scheduled CI run, and
# an unpinned breaking release could silently take down delivery or a source.
# ~= allows patch/bugfix updates but not a new minor. Bump deliberately, then
# re-run the daily workflow manually.

# Required only when TELEGRAPH_ACCESS_TOKEN is set: writes the hosted CV bundle
# with WinZip AES-256. Imported lazily so the stdlib scraper path stays usable
# without optional packages.
pyzipper~=0.4.0

# Optional: enables jobspy, linkedin-global, and linkedin-israel sources.
# 1.2 replaced tls-client with curl_cffi and `linkedin_fetch_description` with
# `fetch_description`. NOTE (Pi): the vendored ARMv6 tls-client .so under vendor/
# only serves jobspy 1.1.x; the Pi's default LinkedIn source is the stdlib
# linkedin-guest, so do not `pip install -U` jobspy there without a curl_cffi
# build for ARMv6.
python-jobspy~=1.3.0
# Optional: enables the secrettelaviv source (Cloudflare-fronted, needs a real
# browser). Requires `python -m playwright install chromium` after pip install.
playwright~=1.61.0

# TOML parser for Python 3.9-3.10; Python 3.11+ ships tomllib. Keep TOML 1.0
# parity with the standard-library parser used by newer runners.
tomli>=2.0,<2.4; python_version < "3.11"
